```python
import subprocess
import sys

def _ensure_pkg(name, attr=None):
    """检查并安装依赖包，返回模块或指定属性"""
    try:
        mod = __import__(name)
    except ImportError:
        subprocess.check_call(
            [sys.executable, "-m", "pip", "install", name, "--break-system-packages"],
            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
        )
        mod = __import__(name)
    return getattr(mod, attr) if attr else mod

requests = _ensure_pkg("requests")
tqdm = _ensure_pkg("tqdm", "tqdm")
```

缺失 `requests` 或 `tqdm` 时会自动 `pip install --break-system-packages`，无需手动安装。
