            temp_res.append(extracted)

        if insert_list and isinstance(insert_list, list):

                            merged_val = int(merged_val)
                    except (TypeError, ValueError):
                        merged_val = str(left) + str(right)
                    merged_data.append(row[:-2] + [merged_val])
            return merged_data

        # 6. 返回处理后的数据（默认不合并，merge_last_two=True 时合并最后两列）
        return _merge_last_two(replaced) if merge_last_two else replaced

                        width += 2
                    else:
                        width += 1.1
                col_widths_chars[i] = max(col_widths_chars[i], width)

        # 根据总字符宽度动态计算画布宽度，高度根据行数计算
        total_char_width = sum(col_widths_chars)
        fig_width = max(total_char_width * 0.18, 6)  # 0.18 为经验系数
        fig_height = max(rows * 0.5, 2)

        # 计算每列占总宽度的比例
        rel_widths = [w / total_char_width for w in col_widths_chars]

        fig, ax = plt.subplots(figsize=(fig_width, fig_height))
        ax.axis('tight'); ax.axis('off')

        # 绘制表格，传入计算好的 colWidths
        table = ax.table(cellText=data, loc='center', cellLoc='center', colWidths=rel_widths)
        table.auto_set_font_size(False)
        table.set_fontsize(10)
        table.scale(1, 1.8)  # 纵向拉伸以增加行高

        colors = {"蓝色": {"h_bg": "#4472C4", "h_fg": "white", "even": "#D9E1F2"},
                  "绿色": {"h_bg": "#70AD47", "h_fg": "white", "even": "#E2EFDA"},
                  "灰色": {"h_bg": "#7B7B7B", "h_fg": "white", "even": "#EDEDED"},
                  "简约": {"h_bg": "#F2F2F2", "h_fg": "black", "even": "#FFFFFF"}}
        cfg = colors.get(table_style, colors["灰色"])
        for (r, c), cell in table.get_celld().items():
            if r == 0:
                cell.set_facecolor(cfg["h_bg"]); cell.get_text().set_color(cfg["h_fg"]); cell.get_text().set_weight('bold')
            else: cell.set_facecolor(cfg["even"] if r % 2 == 0 else "white")
            cell.set_edgecolor("#D0D0D0")
        plt.savefig(output_path, bbox_inches='tight', dpi=300); plt.close()

    # 执行流程
    raw = _fetch_data()
    processed = _process_data(raw)
    if processed: _generate_image(processed)
    return processed
