    """
    title: 飞书多维表格批量更新记录
    description: 通过飞书开放平台API，批量更新多维表格中指定记录的%订单完成时间%和%存在服务单%字段，输入%id_list%、%order_finish_time_list%、%service_order_list%需一一对应，返回%result%更新结果。时间只保留年月日。
    inputs:
        - app_id (str): 飞书应用ID，eg: "cli_a729a2469afed00c"
        - app_secret (str): 飞书应用密钥，eg: "cPVQ***REDACTED***"
        - app_token (str): 多维表格Token，eg: "PAitb***REDACTED***"
        - table_id (str): 数据表ID，eg: "tbl9bhtIHzww9909"
        - id_list (list): 记录ID列表，eg: ["recvqPmpq5Ce4M", "recvqPu6cPNC5J"]
        - order_finish_time_list (list): 完成时间列表，eg: ["2026-07-30 17:26:11", "2026-07-30 09:15:00"]
        - service_order_list (list): 服务单标记列表，eg: ["是", "否"]
    outputs:
        - result (str): 更新结果信息，eg: "成功更新 2 条记录"
    """

    # 1. 检查输入有效性
    if not (len(id_list) == len(order_finish_time_list) == len(service_order_list)):

    # 2. 时间只保留年月日
    def _extract_date(raw):
        # 兼容 "完成时间：2025-07-11 15:18:43）" 等混杂格式，精准提取 YYYY-MM-DD
        m = re.search(r"\d{4}-\d{2}-\d{2}", str(raw))
        return m.group() if m else str(raw).strip()

    # 3. 获取 tenant_access_token
    def _get_token():
        resp = requests.post(
            "https://open.feishu.cn/open-apis/auth/v3/tenant_access_token/internal",
