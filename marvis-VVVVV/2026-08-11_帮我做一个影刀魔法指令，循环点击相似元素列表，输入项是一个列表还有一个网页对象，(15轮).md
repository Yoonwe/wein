try:
    from xbot.app.logging import trace as print
except:
    from xbot import print


import time



def click_elements_by_xpath(xpath, web_obj):
    """循环点击相似元素列表


    title: 循环点击相似元素列表
    description: 传入一条XPath表达式和一个网页对象，查找所有匹配的相似元素并依次点击，每次点击后智能等待页面加载完成，点击失败时跳过并记录，最终返回执行结果汇总文本（含成功/失败计数及失败序号）。自动兼容影刀web_page对象和Selenium浏览器驱动对象。
    inputs:
        xpath: str - XPath定位表达式，用于匹配所有相似元素
        web_obj: Any - 网页对象，支持影刀web_page对象或Selenium浏览器驱动对象
    outputs:
        result: str - 执行结果描述文本，包含成功/失败计数及失败序号
    """
    driver_type = _detect_driver_type(web_obj)


    if driver_type == "selenium":
        return _click_via_selenium(web_obj, xpath)
    return _click_via_xbot(web_obj, xpath)



def _click_via_selenium(driver, xpath):
    """Selenium路径：find_elements 获取列表，逐个点击"""
    from selenium.webdriver.common.by import By
    from selenium.webdriver.support.ui import WebDriverWait


    elements = driver.find_elements(By.XPATH, xpath)
    total = len(elements)
    success_count = 0
    failed_indices = []


    for i, element in enumerate(elements):
        try:
            element.click()
            success_count += 1
            try:
                WebDriverWait(driver, 10).until(
                    lambda d: d.execute_script("return document.readyState") == "complete"
                )
            except Exception:
                time.sleep(1)
        except Exception as e:
            failed_indices.append(i + 1)
            print(f"[跳过] 第 {i+1} 个元素点击失败，错误: {e}")


    return _build_result(total, success_count, failed_indices)



def _click_via_xbot(web_page, xpath):
    """影刀路径：execute_js 获取数量，用索引XPath逐个点击"""
    # 转义 xpath 中的引号，防止 JS 字符串断裂
    xpath_escaped = xpath.replace("\\", "\\\\").replace("'", "\\'")


    # 获取匹配元素数量
    count_js = (
        "var result = document.evaluate("
        f"'{xpath_escaped}', document, null, "
        "XPathResult.ORDERED_NODE_SNAPSHOT_TYPE, null);"
        "result.snapshotLength;"
    )
    try:
        total = int(web_page.execute_js(count_js))
    except Exception as e:
        return f"获取相似元素数量失败，错误: {e}"


    if total == 0:
        return "执行完成：未找到匹配的相似元素。"


    success_count = 0
    failed_indices = []


    for i in range(1, total + 1):
        indexed_xpath = f"({xpath})[{i}]"
        try:
            web_page.click_element_by_xpath(indexed_xpath)
            success_count += 1
            _wait_page_ready(web_page, "xbot")
        except Exception as e:
            failed_indices.append(i)
            print(f"[跳过] 第 {i} 个元素点击失败，错误: {e}")


    return _build_result(total, success_count, failed_indices)



def _build_result(total, success_count, failed_indices):
    """构建执行结果文本"""
    failed_count = total - success_count
    result = f"执行完成：共找到 {total} 个相似元素，成功点击 {success_count} 个，失败 {failed_count} 个。"
    if failed_indices:
        result += f"\n失败元素序号：{failed_indices}"
    return result



def _detect_driver_type(web_obj):
    """检测网页对象类型：Selenium驱动 / 影刀web_page"""
    if hasattr(web_obj, "find_element") and not hasattr(web_obj, "click_element_by_xpath"):
        return "selenium"
    return "xbot"



def _wait_page_ready(web_obj, driver_type, timeout=10):
    """智能等待页面加载完成"""
    if driver_type == "selenium":
        try:
            from selenium.webdriver.support.ui import WebDriverWait
            WebDriverWait(web_obj, timeout).until(
                lambda d: d.execute_script("return document.readyState") == "complete"
            )
        except Exception:
            time.sleep(1)
    else:
        try:
            if hasattr(web_obj, "wait_page_load"):
                web_obj.wait_page_load(timeout)
            elif hasattr(web_obj, "execute_js"):
                for _ in range(timeout * 2):
                    try:
                        state = web_obj.execute_js("return document.readyState")
                        if state == "complete":
                            break
                    except Exception:
                        pass
                    time.sleep(0.5)
            else:
                time.sleep(0.5)
        except Exception:
            time.sleep(0.5)
